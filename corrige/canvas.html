Dessin statique (exercices 1 et 2 de td/canvas.html)

const canvas = document.getElementById("monCanvas");
const ctx = canvas.getContext("2d");

// Exercice 1 : un cercle centré
ctx.beginPath();
ctx.arc(200, 200, 100, 0, 2 * Math.PI);
ctx.strokeStyle = "#2f5fd9";
ctx.lineWidth = 4;
ctx.stroke();

// Exercice 2 : cercles concentriques, opacité progressive
const nbCercles = 8;
const maxRayon = 180;
const ecart = maxRayon / nbCercles;

for (let i = 1; i <= nbCercles; i++) {
  const rayonActuel = maxRayon - i * ecart; // rayon décroissant
  ctx.globalAlpha = i / nbCercles;          // opacité 0→1 vers le centre
  ctx.beginPath();
  ctx.arc(200, 200, rayonActuel, 0, 2 * Math.PI);
  ctx.strokeStyle = "#2f5fd9";
  ctx.lineWidth = 4;
  ctx.stroke();
}
ctx.globalAlpha = 1; // réinitialisation après la boucle

Animation (exercices 1 et 2 de td/canvas-animation.html)

const canvas = document.getElementById("monCanvas");
const ctx = canvas.getContext("2d");

// Exercice 1 : rayon qui pulse
let rayon = 180;
let sens = -1;

function animer() {
  ctx.clearRect(0, 0, 400, 400);      // Q1
  rayon += sens * 2;                  // Q2
  if (rayon <= 20 || rayon >= 180) sens = -sens;
  dessinerCercle(rayon, 1);           // Q3
  requestAnimationFrame(animer);
}
animer();

// Exercice 2 : écho de cercles
const nbCercles = 6;
let temps = 0;

function lancerAnimation() {
  ctx.clearRect(0, 0, 400, 400);      // Q4
  temps += 0.03;
  for (let i = 0; i < nbCercles; i++) {
    const rayon = 60 + 60 * Math.sin(temps - i * 0.5); // Q5
    const alpha = 1 - i / nbCercles;
    dessinerCercle(rayon, alpha);
  }
  requestAnimationFrame(lancerAnimation);
}
lancerAnimation(); // Q6 : lancement initial dès le chargement
